  Serial.begin(115200);

  config.ledc_channel = LEDC_CHANNEL_0;
  config.ledc_timer   = LEDC_TIMER_0;
  config.pin_d0 = Y2_GPIO_NUM;
  config.pin_d1 = Y3_GPIO_NUM;
  config.pin_d2 = Y4_GPIO_NUM;
  config.pin_d3 = Y5_GPIO_NUM;
  config.pin_d4 = Y6_GPIO_NUM;
  config.pin_d5 = Y7_GPIO_NUM;
  config.pin_d6 = Y8_GPIO_NUM;
  config.pin_d7 = Y9_GPIO_NUM;
  config.pin_xclk = XCLK_GPIO_NUM;
  config.pin_pclk = PCLK_GPIO_NUM;
  config.pin_vsync = VSYNC_GPIO_NUM;
  config.pin_href = HREF_GPIO_NUM;
  config.pin_sccb_sda = SIOD_GPIO_NUM;
  config.pin_sccb_scl = SIOC_GPIO_NUM;
  config.pin_pwdn = PWDN_GPIO_NUM;
  config.pin_reset = RESET_GPIO_NUM;
  config.xclk_freq_hz = 20000000;
  config.pixel_format = PIXFORMAT_JPEG;
  config.frame_size   = FRAMESIZE_VGA;
  config.jpeg_quality = 12;
  config.fb_count     = 2;

  esp_err_t err = esp_camera_init(&config);
  if (err != ESP_OK) {
    Serial.printf("Camera init failed: 0x%x\n", err);
    return;
  }

  WiFi.config(local_IP, gateway, subnet);
  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
  }

  webSocket.begin();
  Serial.println("ORION-1 CAM Online at 192.168.4.50:80");
}

void loop() {
  webSocket.loop();
  camera_fb_t *fb = esp_camera_fb_get();
  if (fb) {
    webSocket.broadcastBIN(fb->buf, fb->len);
    esp_camera_fb_return(fb);
  }
}

AsyncWebServer server(80);
AsyncWebSocket ws("/ws");
Servo radarServo;
DHT dht(PIN_DHT, DHT11);

void handleWebSocketMessage(void *arg, uint8_t *data, size_t len) {
  AwsFrameInfo *info = (AwsFrameInfo*)arg;
  if (info->final && info->index == 0 && info->len == len && info->opcode == WS_TEXT) {
    data[len] = 0;
    String command = (char*)data;
    
    if (command == "W") { /* Drive Forward */ }
    else if (command == "S") { /* Drive Reverse */ }
    else if (command == "A") { /* Strafe Left */ }
    else if (command == "D") { /* Strafe Right */ }
    else if (command == "STOP") { /* Emergency Stop */ }
    else if (command.startsWith("SERVO:")) {
      int angle = command.substring(6).toInt();
      radarServo.write(angle);
    }
  }
}

void setup() {
  Serial.begin(115200);
  pinMode(PIN_HEADLIGHT_1, OUTPUT);
  pinMode(PIN_HEADLIGHT_2, OUTPUT);
  digitalWrite(PIN_HEADLIGHT_1, HIGH);
  digitalWrite(PIN_HEADLIGHT_2, HIGH);

  radarServo.attach(PIN_SERVO_PAN);
  radarServo.write(90);
  dht.begin();

  WiFi.softAP("ESP32_SmartCar", "12345678");
  ws.onEvent(onWsEvent);
  server.addHandler(&ws);
  server.begin();
  Serial.println("ORION-1 Core Mission Control Online: 192.168.4.1");
}
